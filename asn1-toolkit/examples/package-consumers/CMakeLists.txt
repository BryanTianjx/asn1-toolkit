cmake_minimum_required(VERSION 3.16)
project(asn1PackageConsumer LANGUAGES C)
set(CONSUMER_KIND "c" CACHE STRING "c, runtime, compiler, or generated")
if(CONSUMER_KIND STREQUAL "c")
  find_package(asn1Toolkit 0.1 REQUIRED COMPONENTS c_runtime)
  add_executable(consumer c_main.c)
  target_link_libraries(consumer PRIVATE asn1::c_runtime)
  set_target_properties(consumer PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
else()
  enable_language(CXX)
  if(CONSUMER_KIND STREQUAL "runtime")
    find_package(asn1Toolkit REQUIRED COMPONENTS per)
    add_executable(consumer runtime_main.cpp)
    target_link_libraries(consumer PRIVATE asn1::per)
  elseif(CONSUMER_KIND STREQUAL "compiler")
    find_package(asn1Toolkit REQUIRED COMPONENTS frontend semantic codegen)
    add_executable(consumer compiler_main.cpp)
    target_link_libraries(consumer PRIVATE asn1::frontend asn1::semantic asn1::codegen)
  elseif(CONSUMER_KIND STREQUAL "generated")
    find_package(asn1Toolkit REQUIRED COMPONENTS per)
    set(ASN1_HOST_COMPILER "" CACHE FILEPATH "Native asn1cxx; never the target executable")
    if(NOT EXISTS "${ASN1_HOST_COMPILER}")
      message(FATAL_ERROR "Provide native ASN1_HOST_COMPILER")
    endif()
    add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated.cpp" "${CMAKE_CURRENT_BINARY_DIR}/generated.hpp"
      COMMAND "${ASN1_HOST_COMPILER}" --no-emit-cache --codec uper --emit-dir "${CMAKE_CURRENT_BINARY_DIR}"
              "${CMAKE_CURRENT_SOURCE_DIR}/minimal.asn"
      DEPENDS "${ASN1_HOST_COMPILER}" "${CMAKE_CURRENT_SOURCE_DIR}/minimal.asn" VERBATIM)
    add_executable(consumer generated_main.cpp "${CMAKE_CURRENT_BINARY_DIR}/generated.cpp")
    target_include_directories(consumer PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
    target_link_libraries(consumer PRIVATE asn1::per)
  else()
    message(FATAL_ERROR "Unknown CONSUMER_KIND")
  endif()
endif()
# Cross builds compile/link the target consumer; execution is a separate device step.
if(NOT CMAKE_CROSSCOMPILING)
  enable_testing()
  add_test(NAME InstalledConsumer COMMAND consumer)
endif()
